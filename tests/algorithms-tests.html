"""Unit tests for CS 499 Enhancement Two: Algorithms and Data Structures."""

import unittest
from unittest.mock import MagicMock

from Enhanced_AnimalShelter_Algorithms import (
    AnimalShelter,
    DataValidationError,
    MongoConfig,
)


class AnimalShelterAlgorithmsTests(unittest.TestCase):
    def setUp(self):
        self.client = MagicMock()
        self.database = MagicMock()
        self.collection = MagicMock()
        self.client.__getitem__.return_value = self.database
        self.database.__getitem__.return_value = self.collection

        self.shelter = AnimalShelter(
            config=MongoConfig(username="test", password="secret"),
            client=self.client,
            verify_connection=False,
        )

        self.records = [
            {"name": "Zelda", "animal_type": "Dog", "breed": "Labrador"},
            {"name": "Bella", "animal_type": "Cat", "breed": "Domestic Shorthair"},
            {"name": "Charlie", "animal_type": "Dog", "breed": "Beagle"},
            {"name": "Bella", "animal_type": "Dog", "breed": "German Shepherd"},
            {"name": "Max", "animal_type": "Cat", "breed": "Siamese"},
        ]

    def test_merge_sort_orders_records_without_mutating_original(self):
        original_order = [record["name"] for record in self.records]

        sorted_records = self.shelter.merge_sort(self.records, "name")

        self.assertEqual(
            [record["name"] for record in sorted_records],
            ["Bella", "Bella", "Charlie", "Max", "Zelda"],
        )
        self.assertEqual([record["name"] for record in self.records], original_order)

    def test_merge_sort_reverse(self):
        sorted_records = self.shelter.merge_sort(self.records, "name", reverse=True)
        self.assertEqual(
            [record["name"] for record in sorted_records],
            ["Zelda", "Max", "Charlie", "Bella", "Bella"],
        )

    def test_binary_search_returns_all_duplicate_matches(self):
        sorted_records = self.shelter.merge_sort(self.records, "name")

        matches = self.shelter.binary_search(sorted_records, "name", "Bella")

        self.assertEqual(len(matches), 2)
        self.assertTrue(all(record["name"] == "Bella" for record in matches))

    def test_binary_search_returns_empty_list_when_not_found(self):
        sorted_records = self.shelter.merge_sort(self.records, "name")
        self.assertEqual(
            self.shelter.binary_search(sorted_records, "name", "Not Present"),
            [],
        )

    def test_build_lookup_index_groups_records_by_key(self):
        index = self.shelter.build_lookup_index(self.records, "animal_type")

        self.assertEqual(len(index["Dog"]), 3)
        self.assertEqual(len(index["Cat"]), 2)

    def test_frequency_count_summarizes_categories(self):
        counts = self.shelter.frequency_count(self.records, "animal_type")
        self.assertEqual(counts, {"Dog": 3, "Cat": 2})

    def test_read_sorted_uses_custom_algorithm_and_applies_limit_after_sort(self):
        cursor = MagicMock()
        cursor.__iter__.return_value = iter(self.records)
        self.collection.find.return_value = cursor

        rows = self.shelter.read_sorted({}, "name", limit=3)

        self.assertEqual(
            [record["name"] for record in rows],
            ["Bella", "Bella", "Charlie"],
        )
        cursor.limit.assert_not_called()

    def test_algorithms_validate_input(self):
        with self.assertRaises(DataValidationError):
            self.shelter.merge_sort("not a list", "name")
        with self.assertRaises(DataValidationError):
            self.shelter.merge_sort(self.records, "")
        with self.assertRaises(DataValidationError):
            self.shelter.binary_search(self.records, "name", None)


if __name__ == "__main__":
    unittest.main()
