"""Unit tests for CS 499 Milestone Four: Database Enhancement."""

import os
import unittest
from unittest.mock import MagicMock, patch

from Enhanced_AnimalShelter_Database import (
    ASCENDING,
    DESCENDING,
    AnimalShelter,
    DataValidationError,
    MongoConfig,
)


class AnimalShelterDatabaseTests(unittest.TestCase):
    def setUp(self):
        self.client = MagicMock()
        self.database = MagicMock()
        self.collection = MagicMock()
        self.client.__getitem__.return_value = self.database
        self.database.__getitem__.return_value = self.collection

        self.shelter = AnimalShelter(
            config=MongoConfig(),
            client=self.client,
            verify_connection=False,
        )

    def test_build_uri_escapes_credentials(self):
        config = MongoConfig(username="user@example.com", password="p:a/ss")
        uri = config.build_uri()
        self.assertIn("user%40example.com", uri)
        self.assertIn("p%3Aa%2Fss", uri)
        self.assertNotIn("p:a/ss", uri)

    def test_from_environment_loads_configuration_without_hardcoding(self):
        values = {
            "MONGODB_USERNAME": "aacuser",
            "MONGODB_PASSWORD": "secret",
            "MONGODB_HOST": "db.internal",
            "MONGODB_PORT": "27018",
            "MONGODB_DB": "aac",
            "MONGODB_COLLECTION": "animals",
            "MONGODB_TLS": "true",
        }
        with patch.dict(os.environ, values, clear=False):
            config = MongoConfig.from_environment()

        self.assertEqual(config.username, "aacuser")
        self.assertEqual(config.host, "db.internal")
        self.assertEqual(config.port, 27018)
        self.assertTrue(config.tls)

    def test_read_applies_server_side_sort_skip_and_limit(self):
        cursor = MagicMock()
        cursor.sort.return_value = cursor
        cursor.skip.return_value = cursor
        cursor.limit.return_value = cursor
        cursor.__iter__.return_value = iter([{"name": "Bella"}])
        self.collection.find.return_value = cursor

        rows = self.shelter.read(
            {"animal_type": "Dog"},
            sort=[("breed", ASCENDING)],
            skip=5,
            limit=10,
        )

        self.assertEqual(rows, [{"name": "Bella"}])
        self.collection.find.assert_called_once_with({"animal_type": "Dog"}, None)
        cursor.sort.assert_called_once_with([("breed", ASCENDING)])
        cursor.skip.assert_called_once_with(5)
        cursor.limit.assert_called_once_with(10)

    def test_query_validation_blocks_server_side_javascript(self):
        with self.assertRaises(DataValidationError):
            self.shelter.read({"$where": "this.age > 4"})

        with self.assertRaises(DataValidationError):
            self.shelter.read({"$expr": {"$function": {"body": "return true"}}})

    def test_update_rejects_unsupported_operator(self):
        with self.assertRaises(DataValidationError):
            self.shelter.update({"name": "Bella"}, {"$replaceWith": {"name": "X"}})

    def test_empty_mass_write_filters_are_rejected(self):
        with self.assertRaises(DataValidationError):
            self.shelter.update({}, {"status": "reviewed"}, many=True)
        with self.assertRaises(DataValidationError):
            self.shelter.delete({}, many=True)

    def test_count_uses_count_documents(self):
        self.collection.count_documents.return_value = 42
        count = self.shelter.count({"animal_type": "Dog"})
        self.assertEqual(count, 42)
        self.collection.count_documents.assert_called_once_with({"animal_type": "Dog"})

    def test_ensure_indexes_creates_expected_index_models(self):
        self.collection.create_indexes.return_value = [
            "idx_breed",
            "idx_animal_type",
            "idx_animal_type_breed",
            "idx_outcome_type",
        ]

        names = self.shelter.ensure_indexes()

        self.assertEqual(len(names), 4)
        self.collection.create_indexes.assert_called_once()
        models = self.collection.create_indexes.call_args.args[0]
        self.assertEqual(len(models), 4)

    def test_aggregate_by_builds_server_side_pipeline(self):
        self.collection.aggregate.return_value = iter(
            [{"_id": "Dog", "count": 30}, {"_id": "Cat", "count": 20}]
        )

        result = self.shelter.aggregate_by(
            "animal_type", match={"outcome_type": "Adoption"}, limit=5
        )

        self.assertEqual(result[0]["_id"], "Dog")
        pipeline = self.collection.aggregate.call_args.args[0]
        self.assertEqual(pipeline[0], {"$match": {"outcome_type": "Adoption"}})
        self.assertEqual(
            pipeline[1],
            {"$group": {"_id": "$animal_type", "count": {"$sum": 1}}},
        )
        self.assertEqual(pipeline[-1], {"$limit": 5})
        self.collection.aggregate.assert_called_once_with(pipeline, allowDiskUse=True)

    def test_explain_query_returns_plan_and_applies_sort(self):
        cursor = MagicMock()
        cursor.sort.return_value = cursor
        cursor.explain.return_value = {"queryPlanner": {"winningPlan": {"stage": "IXSCAN"}}}
        self.collection.find.return_value = cursor

        plan = self.shelter.explain_query(
            {"breed": "Labrador Retriever Mix"},
            sort=[("breed", DESCENDING)],
        )

        self.assertEqual(plan["queryPlanner"]["winningPlan"]["stage"], "IXSCAN")
        cursor.sort.assert_called_once_with([("breed", DESCENDING)])

    def test_create_preserves_crud_behavior(self):
        self.collection.insert_one.return_value.acknowledged = True
        self.assertTrue(self.shelter.create({"name": "Frank", "animal_type": "Dog"}))


if __name__ == "__main__":
    unittest.main()
