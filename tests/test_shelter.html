"""Portfolio unit tests for the original CS 340 AnimalShelter artifact.

These tests use mocks so the original CRUD behavior can be reviewed without
requiring a live MongoDB server.
"""

import unittest
from unittest.mock import MagicMock, patch

from Original_AnimalShelter import AnimalShelter


class OriginalAnimalShelterTests(unittest.TestCase):
    def setUp(self):
        # Patch MongoClient because the original constructor creates the client
        # internally instead of accepting an injected dependency.
        self.mongo_patcher = patch("Original_AnimalShelter.MongoClient")
        self.mock_mongo_client = self.mongo_patcher.start()
        self.addCleanup(self.mongo_patcher.stop)

        self.client = MagicMock()
        self.database = MagicMock()
        self.collection = MagicMock()

        self.mock_mongo_client.return_value = self.client
        self.client.__getitem__.return_value = self.database
        self.database.__getitem__.return_value = self.collection

        self.shelter = AnimalShelter(
            username="testuser",
            password="testpassword",
            db_name="aac",
            collection_name="animals",
        )

    def test_create_valid_document_returns_true(self):
        self.collection.insert_one.return_value.acknowledged = True

        result = self.shelter.create(
            {"name": "Frank", "animal_type": "Dog"}
        )

        self.assertTrue(result)
        self.collection.insert_one.assert_called_once_with(
            {"name": "Frank", "animal_type": "Dog"}
        )

    def test_create_rejects_empty_document(self):
        self.assertFalse(self.shelter.create({}))
        self.collection.insert_one.assert_not_called()

    def test_read_returns_records_and_applies_limit(self):
        cursor = MagicMock()
        cursor.limit.return_value = cursor
        cursor.__iter__.return_value = iter(
            [{"name": "Bella"}, {"name": "Charlie"}]
        )
        self.collection.find.return_value = cursor

        rows = self.shelter.read(
            {"animal_type": "Dog"},
            limit=2,
        )

        self.assertEqual(
            rows,
            [{"name": "Bella"}, {"name": "Charlie"}],
        )
        cursor.limit.assert_called_once_with(2)

    def test_update_wraps_plain_values_in_set_operator(self):
        self.collection.update_one.return_value.modified_count = 1

        modified = self.shelter.update(
            {"name": "Frank"},
            {"age_upon_outcome": "7 years"},
        )

        self.assertEqual(modified, 1)
        self.collection.update_one.assert_called_once_with(
            {"name": "Frank"},
            {"$set": {"age_upon_outcome": "7 years"}},
        )

    def test_delete_rejects_empty_query(self):
        deleted = self.shelter.delete({})

        self.assertEqual(deleted, 0)
        self.collection.delete_one.assert_not_called()
        self.collection.delete_many.assert_not_called()

    def test_delete_returns_deleted_count(self):
        self.collection.delete_one.return_value.deleted_count = 1

        deleted = self.shelter.delete({"name": "Frank"})

        self.assertEqual(deleted, 1)
        self.collection.delete_one.assert_called_once_with(
            {"name": "Frank"}
        )


if __name__ == "__main__":
    unittest.main()
