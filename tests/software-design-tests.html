"""Unit tests for the CS 499 Software Design and Engineering enhancement."""

import unittest
from unittest.mock import MagicMock

from Enhanced_AnimalShelter import AnimalShelter, DataValidationError, MongoConfig


class AnimalShelterSoftwareDesignTests(unittest.TestCase):
    def setUp(self):
        self.client = MagicMock()
        self.database = MagicMock()
        self.collection = MagicMock()
        self.client.__getitem__.return_value = self.database
        self.database.__getitem__.return_value = self.collection

        self.shelter = AnimalShelter(
            config=MongoConfig(username="test", password="secret"),
            client=self.client,
            verify_connection=False,
        )

    def test_create_rejects_empty_document(self):
        with self.assertRaises(DataValidationError):
            self.shelter.create({})

    def test_delete_rejects_empty_query(self):
        with self.assertRaises(DataValidationError):
            self.shelter.delete({})

    def test_update_wraps_plain_fields_in_set_operator(self):
        result = MagicMock(modified_count=1)
        self.collection.update_one.return_value = result

        modified = self.shelter.update(
            {"name": "Frank"},
            {"age_upon_outcome": "7 years"},
        )

        self.assertEqual(modified, 1)
        self.collection.update_one.assert_called_once_with(
            {"name": "Frank"},
            {"$set": {"age_upon_outcome": "7 years"}},
        )

    def test_read_retains_original_limit_behavior(self):
        cursor = MagicMock()
        cursor.limit.return_value = cursor
        cursor.__iter__.return_value = iter([{"name": "Frank"}])
        self.collection.find.return_value = cursor

        rows = self.shelter.read({"animal_type": "Dog"}, limit=5)

        self.assertEqual(rows, [{"name": "Frank"}])
        cursor.limit.assert_called_once_with(5)

    def test_injected_client_is_not_closed_by_shelter(self):
        self.shelter.close()
        self.client.close.assert_not_called()


if __name__ == "__main__":
    unittest.main()
